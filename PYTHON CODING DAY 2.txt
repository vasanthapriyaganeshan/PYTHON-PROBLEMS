1)Find Missing Number

class Solution:
    def missingNumber(self, nums):
        n = len(nums)

        total = n * (n + 1) // 2
        arr_sum = sum(nums)

        return total - arr_sum

2)Reverse an Array

class Solution:
    def reverseArray(self, arr):
        left = 0
        right = len(arr) - 1

        while left < right:
            arr[left], arr[right] = arr[right], arr[left]
            left += 1
            right -= 1

        return arr

3)Remove Element

class Solution:
    def removeElement(self, nums, val):
        k = 0

        for i in range(len(nums)):
            if nums[i] != val:
                nums[k] = nums[i]
                k += 1

        return k

4)Remove Duplicates From Sorted Array

class Solution:
    def removeDuplicates(self, nums):
        k = 1

        for i in range(1, len(nums)):
            if nums[i] != nums[k - 1]:
                nums[k] = nums[i]
                k += 1

        return k

5)Move Zeroes

class Solution:
    def moveZeroes(self, nums):
        k = 0

        for i in range(len(nums)):
            if nums[i] != 0:
                nums[k] = nums[i]
                k += 1

        for i in range(k, len(nums)):
            nums[i] = 0
